cmake_minimum_required(VERSION 3.16)
project(HeapTreapMerkle LANGUAGES CXX)

add_library(heap_headers INTERFACE)
target_compile_features(heap_headers INTERFACE cxx_std_17)
target_include_directories(heap_headers INTERFACE heap/include)
if(MSVC)
    target_compile_options(heap_headers INTERFACE /W4 /utf-8)
else()
    target_compile_options(heap_headers INTERFACE -Wall -Wextra -Wpedantic)
endif()

add_executable(heap_benchmark heap/benchmark.cpp)
target_link_libraries(heap_benchmark PRIVATE heap_headers)

option(BUILD_COURSEWORK_EXPERIMENTS "Build the standalone heap coursework experiments" OFF)
if(BUILD_COURSEWORK_EXPERIMENTS)
    find_package(Threads REQUIRED)
    file(GLOB experiments CONFIGURE_DEPENDS heap/experiments/*.cpp heap/bonus/*.cpp)
    foreach(source IN LISTS experiments)
        get_filename_component(target "${source}" NAME_WE)
        add_executable("${target}" "${source}")
        target_link_libraries("${target}" PRIVATE heap_headers Threads::Threads)
    endforeach()
endif()
